import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
import { readFileSync, readdirSync } from 'fs'
import { join, resolve } from 'path'
import { render, screen, fireEvent, within } from '@testing-library/react'
import { OrganizationView, type OrganizationViewProps } from '@/components/OrganizationView'
import { DEFAULT_ORG_BACKLOG_SETTINGS } from '@/simulation/organizationEngine'
import type { OrgSimState } from '@/types/organization'
import { orgT, runUntil, stateOf } from './feat028Fixtures'

// feat-028 (Týmy složené z lidí) — dlouhé názvy (J4) a barvy jen z tokenů (J10).
//
// Schválený kontrakt (analýza J4, J10; feat-023 pro zkracování):
//  - Název týmu (pole „Team name“ před během, text za běhu) a název oblasti v nabídce mandátů se zkracují výpustkou;
//    plný název je v `title`, jen když je text opravdu zkrácený (`scrollWidth > clientWidth`, jako feat-023).
//  - CSS moduly nových komponent módu Organization (soubory `Org*.module.css`, `Organization*.module.css`,
//    `RangeSlider.module.css`) nepoužívají natvrdo zapsané barvy — jen tokeny z `globals.css` (var(--…)), takže
//    fungují ve světlém i tmavém tématu; kontrast tokenů hlídají existující testy feat-021.

const LONG_TEAM = 'Customer Experience and Onboarding Squad'
const LONG_AREA = 'Customer Experience Operations Platforms'

beforeEach(() => {
  // jsdom nepočítá rozložení: text s „Customer Experience“ je širší než prvek, ostatní se vejde (vzor feat-023)
  vi.spyOn(Element.prototype, 'scrollWidth', 'get').mockImplementation(function (this: Element) {
    const text = this instanceof HTMLInputElement ? this.value : this.textContent ?? ''
    return text.includes('Customer Experience') ? 300 : 50
  })
  vi.spyOn(Element.prototype, 'clientWidth', 'get').mockReturnValue(100)
  Element.prototype.scrollIntoView = vi.fn()
})

afterEach(() => {
  vi.restoreAllMocks()
})

/** Vykreslí plochu módu nad daným stavem. */
function renderView(state: OrgSimState, locked: boolean) {
  const props: OrganizationViewProps = {
    setup: state.setup, onSetupChange: vi.fn(), state, locked,
    backlogSettings: DEFAULT_ORG_BACKLOG_SETTINGS, onBacklogSettingsChange: vi.fn(), onGenerateBacklog: vi.fn(),
    wipMode: 'reduce-wip', onWipModeChange: vi.fn(), revenueProfile: 'flat', onRevenueProfileChange: vi.fn(),
    previousRun: null,
  }
  return render(<OrganizationView {...props} />)
}

/** ORG-T s dlouhým názvem Team Web a dlouhým názvem oblasti BIL. */
function longState(): OrgSimState {
  const s = stateOf(orgT(), 'F-1 (p1): WEB [Frontend 3]')
  s.setup.teams[2].name = LONG_TEAM
  s.setup.areas[0].name = LONG_AREA
  return s
}

describe('feat-028 J4: dlouhé názvy', () => {
  it('před během má pole názvu dlouhého týmu title s plným názvem, krátký název title nemá', () => {
    renderView(longState(), false)
    const inputs = screen.getAllByRole('textbox', { name: 'Team name' })
    expect(inputs[2]).toHaveAttribute('title', LONG_TEAM)
    expect(inputs[0]).not.toHaveAttribute('title')
  })

  it('nabídka mandátů ukáže dlouhý název oblasti s title', () => {
    renderView(longState(), false)
    const row = screen.getAllByTestId('org-team-row')[2]
    fireEvent.click(within(row).getByRole('button', { name: 'Add mandate' }))
    const item = within(screen.getByRole('menu')).getByRole('menuitem', { name: new RegExp(LONG_AREA) })
    expect(item.querySelector(`[title="${LONG_AREA}"]`) ?? item).toHaveAttribute('title', LONG_AREA)
  })

  it('za běhu má zkrácený název týmu title s plným názvem', () => {
    renderView(runUntil(longState(), 0.5), true)
    expect(screen.getByText(LONG_TEAM)).toHaveAttribute('title', LONG_TEAM)
    expect(screen.getByText('Team Billing')).not.toHaveAttribute('title')
  })
})

describe('feat-028 J10: barvy nových komponent jen z tokenů', () => {
  const dir = resolve(__dirname, '../../../src/components')
  const files = readdirSync(dir).filter(n => /^(Org|Organization|RangeSlider).*\.module\.css$/.test(n))
  // Natvrdo zapsaná barva: hex, rgb()/rgba(), hsl()/hsla(), oklch(), oklab(), lab(), lch()
  const LITERAL_COLOR = /#[0-9a-fA-F]{3,8}\b|\b(?:rgba?|hsla?|oklch|oklab|lab|lch)\(/

  it('mód má vlastní CSS moduly (jinak by hlídač nic nehlídal)', () => {
    expect(files.length).toBeGreaterThan(0)
    expect(files).toContain('RangeSlider.module.css')
  })

  it('žádný z nich nemá natvrdo zapsanou barvu', () => {
    const offenders = files.flatMap(name => {
      const css = readFileSync(join(dir, name), 'utf8').replace(/\/\*[\s\S]*?\*\//g, '')
      return css.split('\n').filter(line => LITERAL_COLOR.test(line)).map(line => `${name}: ${line.trim()}`)
    })
    expect(offenders).toEqual([])
  })
})
