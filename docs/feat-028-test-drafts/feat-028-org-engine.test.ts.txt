// feat-028 (Týmy složené z lidí) — engine módu Organization: přidělování práce, Waiting a Idle, metriky,
// výnos, timeline hotové featury, Reset.
//
// Schválený kontrakt (spec features/feat-028-tymy-slozene-z-lidi.md, „Pravidla chování“; testovací analýza
// docs/feat-028-test-analysis.md, skupiny D, E, F, G, I3 a J):
//  - `makeOrgState(setup, backlog)` → stav před během; `tickOrg(state, dt, wipMode)` posune běh o `dt` pracovních dní
//    (smí stav měnit na místě a vrací ho). Engine je deterministický (bez náhody).
//  - Úkol smí vzít člověk se skillem úkolu, jehož tým má mandát na oblast úkolu; člověk pracuje na jednom úkolu.
//  - Pořadí přidělování v jednom okamžiku: týmy shora dolů, lidé v týmu shora dolů; ve featuře úkoly v pořadí oblastí
//    a úkolů; vyšší priorita (nižší číslo) má přednost.
//  - Fáze platí v rámci oblasti: úkol fáze N čeká jen na nižší fáze v téže oblasti téže featury (R8).
//  - Reduce WIP: rozpracované featury před backlogem. Min units: 1) featura, na které pracoval někdo z mého týmu,
//    2) featura, na které nepracoval nikdo z jiného týmu, 3) priorita (R10).
//  - AI 2× / 5× u dvojice člověk + skill zrychlí práci na úkolu tohoto skillu (feat-019).
//  - Úkol si vede úseky práce `segments` (od, do, člověk, tým); featura `startedAt` (první přidělení) a `finishedAt`.
//  - `state.idleByPerson[personId]`: Idle time člověka — roste, když nemá úkol a existuje nepřidělený úkol s jeho
//    skillem kdekoli (bez ohledu na fázi a mandát, R1).
//  - `orgTaskStatus(state, feature, task)`: 'done' | 'doing' | 'todo' | 'waiting' (R2: todo, fáze odemčená,
//    rozpracovaná featura, existuje idle člověk se skillem, jehož tým nemá mandát na oblast).
//  - `orgPersonStatus(state, personId)`: { kind: 'working', featureId, taskId } | { kind: 'idle' } |
//    { kind: 'blocked', areas } (žlutý idle: odemčená práce s jeho skillem čeká v oblasti mimo mandát jeho týmu,
//    v backlogu i v rozpracovaných featurách; `areas` = kódy těch oblastí).
//  - `orgMetrics(state)`: totalTime, avgCycleTime (null bez hotových featur), avgWip (null v čase 0), idleTime,
//    revenue (součet vydělaného výnosu), doneCount.
//  - Výnos jako Cash Flow: hotová featura připíše `revenuePerTick` každých 5 pracovních dní od dokončení
//    (`totalRevenue`, `lastRevenueTickAt`).
//  - `featureTimeline(feature, setup)`: hlavička (startedAt, finishedAt, cycleTime), oblasti s týmy v pořadí prvního
//    začátku práce (shoda → pořadí v panelu, R3), řádky úkolů s čekáním od začátku featury do začátku práce a úseky
//    práce, součty workTotal / waitingTotal a pořadí týmů `teams`.
//  - `resetOrgState(state)` vrátí backlog do stavu před během (stejné featury, priority, výnosy) a nechá organizaci;
//    `snapshotOrgRun(state)` uloží doběhnutý běh pro srovnání.
//
// Časy se porovnávají s tolerancí ±0.05 dne (toBeCloseTo(x, 1)), krok simulace je 0.01 dne.

/** Shodný úsek práce `[jméno, od, do]` s tolerancí ±0.05 dne. */
function expectSegments(actual: Array<[string, number, number | null]>, expected: Array<[string, number, number]>): void {
  expect(actual.map(s => s[0])).toEqual(expected.map(s => s[0]))
  actual.forEach(([, from, to], i) => {
    expect(from).toBeCloseTo(expected[i][1], 1)
    expect(to).not.toBeNull()
    expect(to as number).toBeCloseTo(expected[i][2], 1)
  })
}

/** Cycle time hotové featury. */
function cycleOf(state: OrgSimState, id: number): number {
  const f = featureOf(state, id)
  expect(f.finishedAt).not.toBeNull()
  expect(f.startedAt).not.toBeNull()
  return (f.finishedAt as number) - (f.startedAt as number)
}

describe('feat-028 J3: velká organizace', () => {
  /** 15 týmů po 6 lidech, 8 oblastí; každý tým má všechny 4 skilly a mandát na 1–2 oblasti. */
  function bigSetup(): OrgSetup {
    const codes = ['BIL', 'PAY', 'MOB', 'WEB', 'CRM', 'API', 'DWH', 'IDN']
    const skillCycle = ['Product', 'Java', 'Frontend', 'QA', 'Java', 'QA']
    const setup = makeSetup(Array.from({ length: 15 }, (_, i) => [
      `Team ${i + 1}`,
      i < 8 ? [codes[i]] : [codes[i % 8], codes[(i + 3) % 8]],
      skillCycle.map((sk, j) => [`P${i + 1}-${j + 1}`, [sk]] as [string, string[]]),
    ]))
    setup.areas = codes.map(code => ({ code, name: code }))
    return setup
  }

  it('100 featur doběhne do 2 s strojového času s dt = 0.1 a výsledek je při stejném seedu stejný', () => {
    const runOnce = () => {
      const setup = bigSetup()
      const backlog = generateOrgBacklog(mulberry32(1), DEFAULT_ORG_BACKLOG_SETTINGS, setup, 'flat')
      expect(validateOrgSetup(setup, backlog)).toEqual([])
      let s = makeOrgState(setup, backlog)
      while (!s.finished) s = tickOrg(s, 0.1, 'reduce-wip')
      return orgMetrics(s)
    }
    const t0 = performance.now()
    const first = runOnce()
    const elapsed = performance.now() - t0
    expect(first.doneCount).toBe(100)
    expect(elapsed).toBeLessThan(2000)
    expect(runOnce()).toEqual(first)
  })
})
