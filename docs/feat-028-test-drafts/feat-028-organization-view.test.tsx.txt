/**
 * Vykreslí plochu módu nad daným stavem. Bez `locked` se bere, zda běh začal (čas > 0).
 */
function renderView(state: OrgSimState, overrides: Partial<OrganizationViewProps> = {}) {
  const props: OrganizationViewProps = {
    setup: state.setup,
    onSetupChange: vi.fn(),
    state,
    locked: state.simTime > 0,
    backlogSettings: DEFAULT_ORG_BACKLOG_SETTINGS,
    onBacklogSettingsChange: vi.fn(),
    onGenerateBacklog: vi.fn(),
    wipMode: 'reduce-wip',
    onWipModeChange: vi.fn(),
    revenueProfile: 'flat',
    onRevenueProfileChange: vi.fn(),
    previousRun: null,
    ...overrides,
  }
  return { ...render(<OrganizationView {...props} />), props }
}

const region = (name: string) => screen.getByRole('region', { name })
/** Označení featury v UI jako v ostatních módech: F-001. */
const label = (id: number) => `F-${String(id).padStart(3, '0')}`
const tile = (label: string) => screen.getByTestId(`stat-tile-${label}`)

/** Řádek týmu v panelu organizace podle názvu. */
function teamRow(name: string): HTMLElement {
  const rows = within(region('Organization')).getAllByTestId('org-team-row')
  const row = rows.find(r => within(r).queryByText(name) || within(r).queryByDisplayValue(name))
  if (!row) throw new Error(`Řádek týmu ${name} není`)
  return row
}

/** Rozbalí lidi týmu (pokud ještě nejsou) a vrátí kartu člověka. */
function personCard(teamName: string, personName: string): HTMLElement {
  const row = teamRow(teamName)
  const toggle = within(row).getByRole('button', { name: 'Show or hide people' })
  if (toggle.getAttribute('aria-expanded') !== 'true') fireEvent.click(toggle)
  const card = within(teamRow(teamName)).getAllByTestId('org-person').find(c => within(c).queryByText(personName))
  if (!card) throw new Error(`Karta ${personName} není`)
  return card
}

/** Řádek rozpracované featury. */
function featureRow(id: number): HTMLElement {
  const row = within(region('In progress')).getAllByTestId('org-feature-row').find(r => r.dataset.featureId === String(id))
  if (!row) throw new Error(`F-${id} v In progress není`)
  return row
}

/** Rozbalí řádek rozpracované featury a vrátí jeho úkoly. */
function expandFeature(id: number): HTMLElement[] {
  const toggle = within(featureRow(id)).getByRole('button', { name: new RegExp(`^${label(id)}\\b`) })
  if (toggle.getAttribute('aria-expanded') !== 'true') fireEvent.click(toggle)
  return within(featureRow(id)).getAllByTestId('org-task')
}

/** Štítky oblastí řádku featury jako `[text, stav]`. */
function areaTags(id: number): Array<[string, string | undefined]> {
  return within(featureRow(id)).getAllByTestId('org-area-tag').map(t => [t.textContent?.trim() ?? '', t.dataset.state])
}

/** Hodnota progressbaru v řádku featury. */
const progressOf = (id: number) => within(featureRow(id)).getAllByRole('progressbar')[0].getAttribute('aria-valuenow')

describe('feat-028 F6–F9: srovnání s předchozím během', () => {
  /** Doběhnutý běh: Total 9.0, Avg cycle 6.9, Avg WIP 4.6, Idle 3.0, výnos €450 (jedna featura €450 hotová v 3.0). */
  function finishedRun(): OrgSimState {
    const done = (id: number, startedAt: number, finishedAt: number, rev: number, earned: number): OrgFeature => {
      const f = makeOrgFeature({ id, priority: id, revenuePerTick: rev, areas: [{ area: 'BIL', tasks: [{ role: 'Java', work: 1 }] }] })
      f.tasks.forEach(t => { t.status = 'done'; t.progress = t.work })
      return { ...f, status: 'done', startedAt, finishedAt, totalRevenue: earned, revenueTickCount: earned ? 1 : 0, lastRevenueTickAt: earned ? finishedAt + 5 : finishedAt }
    }
    const s = makeOrgState(orgB(), [])
    s.done = [done(1, 0, 3, 450, 450), done(2, 0, 9, 100, 0), done(3, 0.3, 9, 100, 0)]
    s.simTime = 9
    s.wipIntegral = 4.6 * 9
    s.idleByPerson = { 1: 3 }
    s.startedAt = 0
    s.finished = true
    return s
  }

  /** Předchozí běh: Total 12.0, Avg cycle 8.7, Avg WIP 4.3, Idle 17.0; výnos €300 v čase 9.0, celkem €600. */
  const previous: OrgRunSnapshot = {
    totalTime: 12, avgCycleTime: 8.7, avgWip: 4.3, idleTime: 17, totalRevenue: 600,
    doneFeatures: [{ finishedAt: 3.5, revenuePerTick: 300 }, { finishedAt: 6, revenuePerTick: 300 }],
  }

  /** Řádek delty pod hodnotou dlaždice. */
  const deltaOf = (label: string) => within(tile(label)).getByText('vs prev run').parentElement!

  it('F6: delty ve formátu Cash Flow s barvou podle toho, zda je změna dobrá', () => {
    renderView(finishedRun(), { previousRun: previous })
    expect(tile('Total Time')).toHaveTextContent('1 week 4.0 days')
    expect(deltaOf('Total Time')).toHaveTextContent('↓ 25%')
    expect(deltaOf('Total Time')).toHaveClass(tileStyles.deltaGood)
    expect(tile('Avg Cycle Time')).toHaveTextContent('6.9')
    expect(deltaOf('Avg Cycle Time')).toHaveTextContent('↓ 21%')
    expect(deltaOf('Avg WIP')).toHaveTextContent('↑ 7.0%')
    expect(deltaOf('Avg WIP')).toHaveClass(tileStyles.deltaBad)
    expect(tile('Revenue (time-matched)')).toHaveTextContent('€450')
    expect(within(tile('Revenue (time-matched)')).getByText('@ 1 week 4.0 days')).toBeInTheDocument()
    expect(deltaOf('Revenue (time-matched)')).toHaveTextContent('↑ 50%')
    expect(deltaOf('Revenue (time-matched)')).toHaveClass(tileStyles.deltaGood)
    expect(tile('Idle Time')).toHaveTextContent('3.0')
    expect(deltaOf('Idle Time')).toHaveTextContent('↓ 82%')
    expect(deltaOf('Idle Time')).toHaveClass(tileStyles.deltaGood)
  })

  it('F6: Details výnosu ukazují vlastní celek a oba údaje předchozího běhu', () => {
    renderView(finishedRun(), { previousRun: previous })
    const revenue = tile('Revenue (time-matched)')
    fireEvent.click(within(revenue).getByRole('button', { name: /Details/ }))
    expect(within(revenue).getByText('Total €450 in 1 week 4.0 days')).toBeInTheDocument()
    expect(within(revenue).getByText('Previous run @ 1 week 4.0 days: €300')).toBeInTheDocument()
    expect(within(revenue).getByText('Previous total €600 in 2 weeks 2.0 days')).toBeInTheDocument()
  })

  it('F7: stejný Avg WIP → „→ 0.0% vs prev run“ šedě', () => {
    renderView(finishedRun(), { previousRun: { ...previous, avgWip: 4.6 } })
    expect(deltaOf('Avg WIP')).toHaveTextContent('→ 0.0%')
    expect(deltaOf('Avg WIP')).toHaveClass(tileStyles.deltaFlat)
  })

  it('F8: bez předchozího běhu nemá žádná dlaždice „vs prev run“', () => {
    renderView(finishedRun(), { previousRun: null })
    expect(screen.queryByText('vs prev run')).toBeNull()
  })

  it('F9: za běhu ukazují delty Avg Cycle Time, Avg WIP a Idle Time, Total Time až po doběhnutí', () => {
    const running = d7At(3.5)
    renderView(running, { previousRun: previous })
    for (const label of ['Avg Cycle Time', 'Avg WIP', 'Idle Time']) {
      expect(within(tile(label)).getByText('vs prev run')).toBeInTheDocument()
    }
    expect(within(tile('Total Time')).queryByText('vs prev run')).toBeNull()
  })
})

describe('feat-028 G1, G5, G6: timeline v Done', () => {
  const d6Done = () => runToEnd(stateOf(orgT(), 'F-1 (p1): BIL [Product 2 · Java 3 · QA 2] | MOB [Frontend 3 · QA 1]'))

  /** Otevře timeline hotové featury kliknutím na její řádek v Done. */
  function openTimeline(id: number): HTMLElement {
    const row = screen.getAllByTestId('org-done-row').find(r => r.textContent?.includes(label(id)))!
    fireEvent.click(within(row).getByRole('button', { name: new RegExp(`${label(id)}\\b`) }))
    return within(row).getByTestId('org-timeline')
  }

  it('G1: hlavička, nadpisy oblastí, osa, součty a pořadí týmů', () => {
    renderView(d6Done())
    const tl = openTimeline(1)
    expect(tl).toHaveTextContent('Started day 0.0 · done day 7.0 · cycle time 7.0 d')
    expect(within(tl).getByText('BIL done by Team Billing')).toBeInTheDocument()
    expect(within(tl).getByText('MOB done by Team Mobile')).toBeInTheDocument()
    for (const axis of ['d0.0', 'd3.5', 'd7.0']) expect(within(tl).getByText(axis)).toBeInTheDocument()
    expect(tl).toHaveTextContent('Task time: work 11.0 d · waiting 10.0 d')
    expect(tl).toHaveTextContent('Teams: Team Billing → Team Mobile')
    // 5 úseků práce a 3 čekání (Java a QA v BIL, QA v MOB)
    const segs = within(tl).getAllByTestId('org-timeline-segment')
    expect(segs.filter(s => s.dataset.kind === 'work')).toHaveLength(5)
    expect(segs.filter(s => s.dataset.kind === 'wait')).toHaveLength(3)
  })

  it('G5: tooltipy úseků — práce „Java · Ben, Team Billing · day 2.0–5.0“, čekání „QA · waiting 5.0 d“', () => {
    renderView(d6Done())
    const tl = openTimeline(1)
    expect(within(tl).getByTitle('Java · Ben, Team Billing · day 2.0–5.0')).toHaveAttribute('data-kind', 'work')
    expect(within(tl).getByTitle('QA · waiting 5.0 d')).toHaveAttribute('data-kind', 'wait')
  })

  it('G6: každá timeline se otevírá a zavírá sama; rozpracované featury timeline nemají', () => {
    const s = runToEnd(stateOf(orgB(), 'F-1 (p1): BIL [QA 3]', 'F-2 (p3): BIL [Java 1 · QA 1]', 'F-3 (p2): BIL [QA 1]'))
    renderView(s)
    openTimeline(1)
    openTimeline(2)
    const row1 = screen.getAllByTestId('org-done-row').find(r => r.textContent?.includes('F-001'))!
    fireEvent.click(within(row1).getByRole('button', { name: /F-001\b/ }))
    expect(within(row1).queryByTestId('org-timeline')).toBeNull()
    const row2 = screen.getAllByTestId('org-done-row').find(r => r.textContent?.includes('F-002'))!
    expect(within(row2).getByTestId('org-timeline')).toBeInTheDocument()
    expect(within(region('In progress')).queryByTestId('org-timeline')).toBeNull()
  })
})
