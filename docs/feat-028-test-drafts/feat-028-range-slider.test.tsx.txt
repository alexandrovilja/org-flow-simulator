import { describe, it, expect, vi } from 'vitest'
import { useState } from 'react'
import { render, screen, fireEvent, within } from '@testing-library/react'
import { RangeSlider } from '@/components/RangeSlider'
import { RoleSettings } from '@/components/RoleSettings'
import type { RoleMeta } from '@/types/simulation'

// feat-028 (Týmy složené z lidí) — posuvník min–max se dvěma jezdci a limit 10 fází ve sdíleném editoru skillů.
//
// Schválený kontrakt (spec, Settings → Backlog a Skills; analýza A6, A7, B15, B15b):
//  - `RangeSlider` (nová komponenta, `src/components/RangeSlider.tsx`) má props `label`, `min`, `max`, `value: [lo, hi]`,
//    `onChange([lo, hi])`, `unit: [jednotné, množné]` a nepovinný `tooltip` (title kontejneru).
//  - Dva `input type="range"` s přístupnými názvy „<label> minimum“ a „<label> maximum“ a atributy min/max.
//  - Hodnota se píše „1–3 areas“; při shodě jezdců „2 areas“; u jedničky jednotné číslo „1 area“.
//  - Jezdci se nepřekříží: minimum se zastaví na maximu a maximum na minimu.
//  - Když `min === max` (jediná možnost), oba jezdci jsou neaktivní.
//  - Editor skillů (`RoleSettings`, sdílený Advanced, Cash Flow i Organization) má fázi v rozsahu 1–10 (dřív 1–5).

/** Řízený posuvník s vlastním stavem, aby šlo ověřit, co se zobrazí po změně. */
function Harness({ initial, min = 1, max = 20, unit = ['day', 'days'] as [string, string], label = 'Effort per skill' }: {
  initial: [number, number]; min?: number; max?: number; unit?: [string, string]; label?: string
}) {
  const [value, setValue] = useState<[number, number]>(initial)
  return <RangeSlider label={label} min={min} max={max} value={value} onChange={setValue} unit={unit} tooltip="Effort of one task in working days (man-days)." />
}

const minInput = (label = 'Effort per skill') => screen.getByRole('slider', { name: `${label} minimum` })
const maxInput = (label = 'Effort per skill') => screen.getByRole('slider', { name: `${label} maximum` })

describe('feat-028 A7: jezdci se nepřekříží', () => {
  it('minimum tažené na 8 se zastaví na 5 → „5 days“; maximum tažené na 3 se zastaví na 5', () => {
    render(<Harness initial={[1, 5]} />)
    expect(screen.getByText('1–5 days')).toBeInTheDocument()
    fireEvent.change(minInput(), { target: { value: '8' } })
    expect(screen.getByText('5 days')).toBeInTheDocument()
    expect(minInput()).toHaveValue('5')
    fireEvent.change(maxInput(), { target: { value: '3' } })
    expect(screen.getByText('5 days')).toBeInTheDocument()
    expect(maxInput()).toHaveValue('5')
  })

  it('onChange dostane dvojici [min, max]', () => {
    const onChange = vi.fn()
    render(<RangeSlider label="Effort per skill" min={1} max={20} value={[1, 5]} onChange={onChange} unit={['day', 'days']} />)
    fireEvent.change(maxInput(), { target: { value: '12' } })
    expect(onChange).toHaveBeenLastCalledWith([1, 12])
    fireEvent.change(minInput(), { target: { value: '3' } })
    expect(onChange).toHaveBeenLastCalledWith([3, 5])
  })
})

describe('feat-028 A6: zápis hodnoty, meze a neaktivní posuvník', () => {
  it('rozsah a atributy min/max odpovídají props', () => {
    render(<Harness label="Skills per area" initial={[1, 3]} max={4} unit={['skill', 'skills']} />)
    expect(screen.getByText('1–3 skills')).toBeInTheDocument()
    expect(minInput('Skills per area')).toHaveAttribute('min', '1')
    expect(maxInput('Skills per area')).toHaveAttribute('max', '4')
  })

  it('shodní jezdci: „2 areas“, u jedničky jednotné číslo „1 area“', () => {
    const { unmount } = render(<Harness label="Areas per feature" initial={[2, 2]} max={3} unit={['area', 'areas']} />)
    expect(screen.getByText('2 areas')).toBeInTheDocument()
    unmount()
    render(<Harness label="Areas per feature" initial={[1, 1]} max={3} unit={['area', 'areas']} />)
    expect(screen.getByText('1 area')).toBeInTheDocument()
  })

  it('s jedinou možností (min = max = 1) jsou oba jezdci neaktivní a ukazují „1 area“', () => {
    render(<Harness label="Areas per feature" initial={[1, 1]} min={1} max={1} unit={['area', 'areas']} />)
    expect(minInput('Areas per feature')).toBeDisabled()
    expect(maxInput('Areas per feature')).toBeDisabled()
    expect(screen.getByText('1 area')).toBeInTheDocument()
  })

  it('tooltip je na kontejneru posuvníku', () => {
    render(<Harness initial={[1, 5]} />)
    expect(screen.getByTitle('Effort of one task in working days (man-days).')).toBeInTheDocument()
  })
})

describe('feat-028 B15 a B15b: fáze 1–10 ve sdíleném editoru skillů', () => {
  /** Editor s jedním skillem ve fázi 1 a vlastním stavem. */
  function SkillsHarness() {
    const [config, setConfig] = useState<Record<string, RoleMeta>>({
      Testing: { label: 'Testing', color: 'var(--role-green)', level: 1, required: false },
      Java: { label: 'Java', color: 'var(--role-blue)', level: 2, required: false },
    })
    return (
      <RoleSettings
        roleConfig={config}
        onChange={(id, updates) => setConfig(c => ({ ...c, [id]: { ...c[id], ...updates } }))}
        onAdd={() => {}}
        onDelete={() => {}}
      />
    )
  }

  /** Řádek editoru se skillem daného názvu (podle hodnoty pole „Skill name“). */
  function rowOf(label: string): HTMLElement {
    const input = screen.getAllByRole('textbox', { name: 'Skill name' }).find(i => (i as HTMLInputElement).value === label)!
    let row = input.parentElement!
    while (!within(row).queryByRole('button', { name: '+' })) row = row.parentElement!
    return row
  }

  it('+ devětkrát zvedne fázi z 1 na 10 a pak je neaktivní', () => {
    render(<SkillsHarness />)
    const row = rowOf('Testing')
    const plus = within(row).getByRole('button', { name: '+' })
    for (let i = 0; i < 9; i++) fireEvent.click(plus)
    expect(within(row).getByText('10')).toBeInTheDocument()
    expect(plus).toBeDisabled()
  })

  it('− u fáze 1 je neaktivní; z fáze 2 jde dolů na 1', () => {
    render(<SkillsHarness />)
    expect(within(rowOf('Testing')).getByRole('button', { name: '−' })).toBeDisabled()
    const javaRow = rowOf('Java')
    fireEvent.click(within(javaRow).getByRole('button', { name: '−' }))
    expect(within(javaRow).getByRole('button', { name: '−' })).toBeDisabled()
  })
})
