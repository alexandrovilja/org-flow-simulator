/** Posune falešný čas o `ms` milisekund uvnitř act(). */
async function advance(ms: number): Promise<void> {
  await act(async () => { vi.advanceTimersByTime(ms) })
}

/** Vykreslí aplikaci a přepne na mód Organization. */
function openOrganization(): void {
  render(<Simulator />)
  fireEvent.click(screen.getByRole('tab', { name: 'Organization' }))
}

const region = (name: string) => screen.getByRole('region', { name })
const org = () => region('Organization')
const tile = (label: string) => screen.getByTestId(`stat-tile-${label}`)
/** Tlačítko běhu v liště podle konce názvu (glyf ▶ / ❙❙ / ✓ před ním). */
const runButton = (label: 'Run' | 'Pause' | 'Done' = 'Run') => screen.getByRole('button', { name: new RegExp(`${label}$`) })
/** Rozbalovací tlačítko sekce Settings. */
const accordion = (name: 'Backlog' | 'Skills' | 'Areas') => screen.getByRole('button', { name })

/** Otevře sekci Settings, pokud ještě není otevřená. */
function openSettings(name: 'Backlog' | 'Skills' | 'Areas'): void {
  if (accordion(name).getAttribute('aria-expanded') !== 'true') fireEvent.click(accordion(name))
}

/** Řádek týmu podle názvu (před během je název v poli „Team name“). */
function teamRow(name: string): HTMLElement {
  const row = within(org()).getAllByTestId('org-team-row').find(r =>
    within(r).queryByDisplayValue(name) || within(r).queryByText(name))
  if (!row) throw new Error(`Řádek týmu ${name} není`)
  return row
}

/** Všechny řádky týmů. */
const teamRows = () => within(org()).getAllByTestId('org-team-row')

/** Názvy týmů v pořadí panelu (před během z polí „Team name“). */
const teamNames = () => teamRows().map(r => (within(r).queryByRole('textbox', { name: 'Team name' }) as HTMLInputElement | null)?.value ?? '')

/** Rozbalí lidi týmu a vrátí karty lidí. */
function people(teamName: string): HTMLElement[] {
  const toggle = within(teamRow(teamName)).getByRole('button', { name: 'Show or hide people' })
  if (toggle.getAttribute('aria-expanded') !== 'true') fireEvent.click(toggle)
  return within(teamRow(teamName)).queryAllByTestId('org-person')
}

/** Karta člověka podle jména. */
function personCard(teamName: string, name: string): HTMLElement {
  const card = people(teamName).find(c => within(c).queryByText(name))
  if (!card) throw new Error(`Karta ${name} v ${teamName} není`)
  return card
}

/** Názvy tlačítek odebrání mandátu v řádku týmu (pořadí štítků). */
const mandateButtons = (teamName: string) =>
  within(teamRow(teamName)).queryAllByRole('button', { name: /^Remove mandate / }).map(b => b.getAttribute('aria-label'))

/** Přidá týmu mandát přes tlačítko + a nabídku. */
function addMandate(teamName: string, item: string): void {
  fireEvent.click(within(teamRow(teamName)).getByRole('button', { name: 'Add mandate' }))
  fireEvent.click(screen.getByRole('menuitem', { name: item }))
}

/** Text varování v panelu organizace (prázdné pole = žádné varování). */
function warnings(): string[] {
  const alert = within(org()).queryByRole('alert')
  return alert ? within(alert).getAllByRole('listitem').map(li => li.textContent ?? '') : []
}

/** Položky backlogu. */
const backlogItems = () => within(region('Backlog')).queryAllByTestId('org-backlog-item')

/** Rozbalí položku backlogu a vrátí ji. */
function expandItem(item: HTMLElement): HTMLElement {
  const toggle = within(item).getAllByRole('button').find(b => b.hasAttribute('aria-expanded'))!
  if (toggle.getAttribute('aria-expanded') !== 'true') fireEvent.click(toggle)
  return item
}

/** Nechá běh doběhnout při 10× (každý krok 1 s = 10 pracovních dní). */
async function runToDone(): Promise<void> {
  fireEvent.click(screen.getByRole('button', { name: '10×' }))
  fireEvent.click(runButton('Run'))
  for (let i = 0; i < 120 && !screen.queryByRole('button', { name: /Done$/ }); i++) await advance(1000)
  expect(runButton('Done')).toBeInTheDocument()
}

/** Zmenší backlog na 10 featur a vygeneruje nový (kratší běh v testech). */
function smallBacklog(): void {
  openSettings('Backlog')
  fireEvent.change(screen.getByRole('slider', { name: 'Backlog size' }), { target: { value: '10' } })
  fireEvent.click(screen.getByRole('button', { name: 'Generate new backlog' }))
}

    expect(['Team Billing', 'Team Mobile', 'Team Web'].map(mandateButtons)).toEqual([
      ['Remove mandate BIL'], ['Remove mandate MOB'], ['Remove mandate WEB'],
    ])
    expect(within(org()).getByText('3 teams · 12 people')).toBeInTheDocument()
  })

  it('Settings: skilly Product 1, Java 2, Frontend 2, QA 3; oblasti BIL Billing, MOB Mobile app, WEB Web portal', () => {
    openOrganization()
    openSettings('Skills')
    const names = screen.getAllByRole('textbox', { name: 'Skill name' }).map(i => (i as HTMLInputElement).value)
    expect(names).toEqual(['Product', 'Java', 'Frontend', 'QA'])

    const areaNames = screen.getAllByRole('textbox', { name: 'Area name' }).map(i => (i as HTMLInputElement).value)
    expect(areaNames).toEqual(['Billing', 'Mobile app', 'Web portal'])

  })

  it('backlog má 100 featur „sorted by revenue“; metriky na začátku; Run je aktivní a bez varování', () => {
    openOrganization()
    expect(backlogItems()).toHaveLength(100)
    expect(within(region('Backlog')).getByText('sorted by revenue')).toBeInTheDocument()
    expect(tile('Total Time')).toHaveTextContent('0.0 days')
    expect(tile('Revenue (time-matched)')).toHaveTextContent('€0')

    expect(runButton()).toBeEnabled()
  })

describe('feat-028 A11: posuvníky se projeví až po vygenerování', () => {
  it('Effort per skill 10–10 backlog nezmění; po „Generate new backlog“ mají všechny úkoly 10 dní', () => {
    openOrganization()
    const firstBefore = backlogItems()[0].textContent
    openSettings('Backlog')
    fireEvent.change(screen.getByRole('slider', { name: 'Effort per skill maximum' }), { target: { value: '10' } })
    fireEvent.change(screen.getByRole('slider', { name: 'Effort per skill minimum' }), { target: { value: '10' } })
    expect(screen.getByText('10 days')).toBeInTheDocument()
    expect(backlogItems()[0].textContent).toBe(firstBefore)

    fireEvent.click(screen.getByRole('button', { name: 'Generate new backlog' }))
    for (const item of backlogItems().slice(0, 5)) {
      const chips = within(expandItem(item)).getAllByText(/ · \d+ d$/).map(c => c.textContent)
      expect(chips.length).toBeGreaterThan(0)
      for (const chip of chips) expect(chip).toMatch(/ · 10 d$/)
    }
  })
})
